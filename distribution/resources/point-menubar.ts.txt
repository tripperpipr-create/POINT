/*---------------------------------------------------------------------------------------------
 *  Point IDE — верхнее меню по образцу JetBrains.
 *
 *  Code-OSS даёт восемь меню: Файл, Правка, Выделение, Вид, Переход, Выполнить,
 *  Терминал, Справка. Половина из них про редактор текста, а не про работу с
 *  проектом: отдельное меню «Выделение» есть только здесь, «Терминал» занимает
 *  верхний уровень целиком ради шести пунктов, зато ни кода, ни рефакторинга,
 *  ни Git в строке нет вовсе — и это при том, что все нужные команды у Point
 *  давно есть, с жетбрейнсовскими сочетаниями клавиш.
 *
 *  Строка приводится к жетбрейнсовской:
 *
 *      Файл · Правка · Вид · Переход · Код · Выполнить · Git · Инструменты · Справка
 *
 *  «Выделение» уходит подменю в «Правку», «Терминал» — в «Инструменты»: ничего
 *  не теряется, а верхний уровень занимают вещи, за которыми ходят каждый день.
 *  Порядок задаётся дробными числами, чтобы не трогать чужие регистрации:
 *  Файл 1, Правка 2, Вид 4, Переход 5, Выполнить 6, Справка 8 — наши встают
 *  между ними.
 *
 *  Подписи короткие: у команд расширения в палитре они с подсказкой вида
 *  «(как Ctrl+K)», и в меню такая строка читалась бы дважды — сочетание клавиш
 *  меню и так показывает своей колонкой.
 *--------------------------------------------------------------------------------------------*/

		const collect = (globalThis as typeof globalThis & { gc?: () => void }).gc;
		if (typeof collect !== 'function') {
			return false;
		}
		collect();
		return true;
	});
	CommandsRegistry.registerCommand('point.openInMainEditor', async (accessor, raw: { fileUri?: UriComponents; line?: number }) => {
		if (!raw?.fileUri) {
			return false;
		}
		const line = Number.isInteger(raw.line) && Number(raw.line) > 0 ? Number(raw.line) : undefined;
		const editorGroupsService = accessor.get(IEditorGroupsService);
		await accessor.get(IEditorService).openEditor({
			resource: URI.revive(raw.fileUri),
			options: {
				pinned: true,
				...(line ? { selection: { startLineNumber: line, startColumn: 1 } } : {}),
			},
		}, editorGroupsService.mainPart.activeGroup);
		editorGroupsService.mainPart.activeGroup.focus();
		return true;
	});
	CommandsRegistry.registerCommand('point.focusAgentHubWindow', accessor => {
		const editorGroupsService = accessor.get(IEditorGroupsService);
		for (const part of editorGroupsService.parts) {
			if (part === editorGroupsService.mainPart) {
				continue;
			}
			const group = part.groups.find(candidate => candidate.editors.some(editor => editor.getName() === 'Агенты Point'));
			if (group) {
				group.focus();
				return true;
			}
		}
		return false;
	});
